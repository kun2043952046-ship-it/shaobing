# ---------- 3) 3D 重定位: 查看输出 ----------
ros2 topic echo /pose_correction --once            # map->odom (frame=map, z 锁 0)
ros2 topic echo /relocalization_3d/fitness --once   # 截断均方 m^2, 失败为 -1
ros2 topic hz /pose_correction                      # 应 ≈ rate_hz (1 Hz)
ros2 topic hz /relocalization_3d/prior_map          # latched, 只发一次 (新订阅也能收到)
ros2 run tf2_ros tf2_echo map odom                  # TF: map->odom 应 ≈ (-4.9, 3.0, -93°)
ros2 run tf2_ros tf2_echo map livox_frame           # TF: 车在地图中的位姿

# 验收判据 (仿真出生点 -5,3,-90°):
#   fitness ≈ 0.01 m^2, 内点 ≈ 1.00, map->odom ≈ (-4.9, 3.0, -93°)
#   车开动后 map->odom 基本不变 (它是 odom 原点=出生点 在地图中的位姿)

# ---------- 4) 3D 重定位: 参数热改 (无需重启) ----------
ros2 param set /relocalization_3d enabled false               # 停用, 发一次恒等修正
ros2 param set /relocalization_3d rate_hz 2.0                 # 配准频率
ros2 param set /relocalization_3d gicp_fitness_max 2.0        # 截断均方门限 (m^2)
ros2 param set /relocalization_3d fitness_inlier_min 0.3      # 内点比例下限
ros2 param set /relocalization_3d source_range_max 20.0       # 范围过滤 (丢无回波远点)
ros2 param set /relocalization_3d correction_max_delta_m 1.0  # 单帧跳变上限 (m)
ros2 param set /relocalization_3d correction_max_speed_mps 2.0 # 跳变速率门限 (m/s)
ros2 param set /relocalization_3d compensate_odom false       # 传感器系点云时才设 true
ros2 param set /relocalization_3d ema_alpha 0.5               # 0.5=平滑, 1=不平滑
ros2 param set /relocalization_3d tf_publish_rate 20.0        # map->odom TF 广播频率
ros2 param set /relocalization_3d correction_z_locked true    # z 锁 0 (2D 定位)
touch /home/lele/aaaaa_mine_NJIT_Vision_Gen2/Navigation/test/test_fangzhen/map.pcd   # 触发地图重载 -> 回到初始化

# ---------- 5) RViz ----------

# 显示项: 灰蓝=先验图 /relocalization_3d/prior_map, 绿=实时点云 /cloud_registered,
#         红箭头=map->odom /pose_correction, 橙=原始点云 /livox/lidar (默认关)

# ---------- 9) 接口一览 ----------
# /livox/lidar              PointCloud2 (livox_frame)  仿真 Mid-360 原始点云
# /cloud_registered         PointCloud2 (odom)         cloud_to_odom.py 输出 (或 FAST-LIO)
# /pose_correction          PoseStamped  (map)         map->odom 修正量
# /relocalization_3d/fitness Float64                   截断均方, 失败 -1
# /relocalization_3d/prior_map PointCloud2 (map)       先验图 (latched)
# /cmd_vel                  Twist                      速度指令入口
# /odom                     Odometry                   diff_drive 里程计
# /scan                     LaserScan                  lidar_2d 切片
# TF: map -> odom (本节点) -> base_footprint (diff_drive) -> ... -> livox_frame
